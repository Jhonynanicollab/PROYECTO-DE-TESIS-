# Búsqueda de hiperparámetros con cross-validation
grid_xgb <- expand.grid(
  max_depth = c(3, 6),

mejores <- data.frame()
set.seed(123)
for (i in 1:nrow(grid_xgb)) {
  params <- list(
    objective        = "reg:squarederror",
    max_depth        = grid_xgb$max_depth[i],
    eta              = grid_xgb$eta[i],
    gamma            = 0,
    colsample_bytree = 0.8,
    min_child_weight = 1,
    subsample        = 0.8
  )
  cv <- xgb.cv(
    params  = params,
    data    = dtrain,
    nrounds = 200,
    nfold   = 5,
    early_stopping_rounds = 10,
    verbose = 0
  )
  
  # Detectar nombre de columna RMSE del evaluation_log (varía entre versiones)
  eval_log <- as.data.frame(cv$evaluation_log)
  rmse_col <- grep("test.*rmse.*mean", names(eval_log), value = TRUE)[1]
  
  # Obtener mejor iteración (fallback si best_iteration es NULL)
  best_iter <- cv$best_iteration
  if (is.null(best_iter) || length(best_iter) == 0) {
    best_iter <- which.min(eval_log[[rmse_col]])
  }
  
  mejores <- rbind(mejores, data.frame(
    max_depth = grid_xgb$max_depth[i],
    eta       = grid_xgb$eta[i],
    nrounds   = best_iter,
    rmse_cv   = eval_log[[rmse_col]][best_iter]
  ))
}
